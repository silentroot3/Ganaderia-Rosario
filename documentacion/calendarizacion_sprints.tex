    \item Metodología: \textbf{Scrum}

    \item Sprint Review: \textbf{viernes de cada semana}
    \item \textbf{Fecha de entrega final:} 30 de noviembre de 2026
    \item \textbf{Buffer:} 1 día (sábado 29 nov) para ajustes imprevistos
\end{itemize}
